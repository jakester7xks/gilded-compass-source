cmake_minimum_required(VERSION 3.24)
project(GildedCompass VERSION 1.0.0 LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use the Visual Studio 2022 x64 C++ toolchain on Windows")
endif()
add_library(compass_core STATIC src/NavigationMath.cpp)
target_include_directories(compass_core PUBLIC include)
target_compile_features(compass_core PUBLIC cxx_std_20)
add_library(GildedCompassHeading SHARED src/native/HeadingBridge.cpp src/native/HeadingThunk.asm)
target_include_directories(GildedCompassHeading PRIVATE bindings)
target_link_libraries(GildedCompassHeading PRIVATE compass_core bcrypt)
target_compile_definitions(GildedCompassHeading PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(GildedCompassHeading PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
target_link_options(GildedCompassHeading PRIVATE /PDBALTPATH:GildedCompassHeading.pdb)
set_target_properties(GildedCompassHeading PROPERTIES PREFIX "" SUFFIX ".asi")
include(CTest)
if(BUILD_TESTING)
  foreach(name DirectionStrip ObjectiveAssociation HeadingSources NavigationMath)
    add_executable(${name}Tests tests/${name}Tests.cpp)
    target_link_libraries(${name}Tests PRIVATE compass_core)
    add_test(NAME ${name} COMMAND ${name}Tests)
  endforeach()
  add_executable(HeadingThunkTests tests/HeadingThunkTests.cpp src/native/HeadingThunk.asm)
  target_compile_features(HeadingThunkTests PRIVATE cxx_std_20)
  add_test(NAME HeadingThunk COMMAND HeadingThunkTests)
  add_executable(HeadingLoadTests tests/HeadingLoadTests.cpp)
  add_dependencies(HeadingLoadTests GildedCompassHeading)
  add_test(NAME HeadingDisabled COMMAND HeadingLoadTests "$<TARGET_FILE:GildedCompassHeading>")
endif()
# Builds stay in the chosen build directory. Nothing installs into the game.
